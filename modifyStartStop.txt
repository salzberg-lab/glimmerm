void  Find_Stop_Codons  (char X [], int T, int Stop [])

//  Set  Stop [0 .. 6]  TRUE  or  FALSE   according to whether
//  X [1 .. T] has a stop codon in the corresponding reading frame.
//  Stop [6]  is always set  FALSE .

  {
   static int  Next [6] [4] =
     {{ 0,  0,  0,  1},     //  0  a, c, g
      { 0,  2,  3,  1},     //  1  t
      { 5,  0,  0,  1},     //  2  tc
      { 4,  0,  0,  1},     //  3  tg
      { 0,  0,  0,  1},     //  4  tga forward stop codon
      { 0,  0,  0,  1},     //  4  tca reverse stop codon
      };
   int  i, State;

   for  (i = 0;  i < 7;  i ++)
     Stop [i] = 0;

   State = 0;
   for  (i = 1;  i <= T;  i ++)
     {
      switch  (tolower (X [i]))
        {
         case  'a' :
           State = Next [State] [0];
           break;
         case  'c' :
           State = Next [State] [1];
           break;
         case  'g' :
           State = Next [State] [2];
           break;
         case  't' :
           State = Next [State] [3];
           break;
         default :
           fprintf (stderr, "Unexpected character %c\n", X [i]);
           State = 0;
        }
      if  (State == 4)
          Stop [i % 3] = TRUE;
      else if  (State > 4)
          Stop [3 + i % 3] = TRUE;
     }

   return;
  }
